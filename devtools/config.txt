#
# Enable 32-bit mode (AArch32)
#
# This file must be copied along with the generated kernel7[l].img
# onto a SD(HC) card with FAT file system, if 32-bit mode is used.
#
# This file also enables the possibility to use the FIQ and two
# displays with the Raspberry Pi 4 and selects the required mode
# for the USB host controller of the Compute Module 4.
#

arm_64bit=0
initial_turbo=0

[pi4]
kernel=kernel7l.img
armstub=armstub7-rpi4.bin
max_framebuffers=2

[cm4]
otg_mode=1

[all]
# pico-gpu: memory for the VideoCore (firmware OpenGL ES needs it)
gpu_mem=128
# pico-gpu: core clock 300 MHz so that SPI0 can run at 75 MHz (300 / 4)
core_freq=300
core_freq_min=300
# pico-gpu: keep HDMI (not composite) as the output when no monitor is
# connected at boot, so that one plugged in later gets a picture; the gpu app
# watches the hot-plug line itself. No overscan borders.
hdmi_force_hotplug=1
disable_overscan=1
